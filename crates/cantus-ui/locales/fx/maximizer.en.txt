# Parameters (cell labels) and what each one does.
fx.max.gain = Drive
fx.max.gain_tip = How hard the result is pushed into the ceiling. The brickwall holds the peaks; everything under them gets louder. Learn sets it for the target.
fx.max.ceiling = Ceiling
fx.max.ceiling_tip = The most the output ever reaches. With true peak on it is measured between the samples too (dBTP).
fx.max.style = Character
fx.max.style_tip = How the brickwall catches a peak and lets it go: Terminus's five engines, each printed as what it does.
fx.max.look = Lookahead
fx.max.look_tip = How far ahead the brickwall sees a peak coming. Adds this much latency.
fx.max.lim_att = Attack
fx.max.lim_att_tip = How soon the brickwall's slow stage takes over from the fast one.
fx.max.lim_rel = Release
fx.max.lim_rel_tip = How the brickwall lets go after an overload. Longer is smoother; shorter is louder and can pump.
fx.max.link_t = Link transients
fx.max.link_t_tip = Stereo link of the fast stage. Below 100 % a one-sided peak is caught on its side alone.
fx.max.link_r = Link release
fx.max.link_r_tip = Stereo link of the slow stage. Keep it high so the recovery stays centred.
fx.max.lim_knee = Knee
fx.max.lim_knee_tip = The brickwall starts bending this many dB under the ceiling. The ceiling itself is never passed.
fx.max.tp = True peak
fx.max.tp_tip = Hold the peaks between the samples too, as a converter or an encoder sees them.
fx.max.clip = Clipper
fx.max.clip_tip = A soft clipper before the brickwall, oversampled 4x: more of each peak is clipped and less limited as it goes up.
fx.max.dc = DC filter
fx.max.dc_tip = Remove a DC offset before the brickwall.
fx.max.bands = Bands
fx.max.bands_tip = How many bands the spectrum is split into before the full band: none, two or three.
fx.max.x1 = Low split
fx.max.x1_tip = Where the low band ends and the next begins. Drag it on the band display.
fx.max.x2 = High split
fx.max.x2_tip = Where the mid band ends and the high band begins.
fx.max.xmode = Crossovers
fx.max.xmode_tip = How the bands are made: Dynamic (no split, no latency, a wire while nothing moves), Minimum phase (a Linkwitz–Riley split) or Linear phase (no phase shift, 48 ms of latency).
fx.max.slope = Slope
fx.max.delta = Delta
fx.max.delta_tip = Hear only what Culmen changes: the input, driven and lined up, minus the output.
fx.max.monitor = Hear
fx.max.dither = Dither
fx.max.dither_tip = TPDF dither for a lower word length: only the last thing before the file should dither. Digital silence stays silent.
fx.max.shape = Noise shaping
fx.max.target = Target
fx.max.target_tip = The loudness the meters judge and Learn aims at.
fx.max.on = On
fx.max.solo = Solo
fx.max.thr = Threshold
fx.max.thr_tip = Above this level the upper ratio works: over 1:1 it squeezes the peaks down, under 1:1 it lifts them further.
fx.max.ratio = Ratio
fx.max.ratio_tip = The upper side's ratio: 4:1 compresses, 1:2 expands upward, 1:1 is off.
fx.max.range = Range
fx.max.range_tip = The most the upper side moves the gain.
fx.max.thr_lo = Lower threshold
fx.max.thr_lo_tip = Below this level the lower ratio works: over 1:1 it lifts quiet parts (upward compression), under 1:1 it pushes them down.
fx.max.ratio_lo = Lower ratio
fx.max.ratio_lo_tip = The lower side's ratio: 2:1 lifts what is under the threshold, 1:2 expands it down, 1:1 is off.
fx.max.range_lo = Lower range
fx.max.range_lo_tip = The most the lower side moves the gain.
fx.max.knee = Knee
fx.max.att = Attack
fx.max.rel = Release
fx.max.band_gain = Band gain
fx.max.band_gain_tip = The band's own level, after its dynamics.
fx.max.place = Stereo
fx.max.place_tip = What the band listens to and moves: both channels linked, the mid only, or the side only.
