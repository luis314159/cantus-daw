# Parameters (knob labels) and what each one does.
fx.lim.gain = Gain
fx.lim.gain_tip = How hard the signal is driven into the ceiling. The limiter holds the peaks; everything under them gets louder.
fx.lim.ceiling = Ceiling
fx.lim.ceiling_tip = The most the output ever reaches. With true peak on it is measured between the samples too (dBTP).
fx.lim.style = Style
fx.lim.look = Lookahead
fx.lim.look_tip = How far ahead the transient stage sees a peak coming, so it can turn down smoothly instead of clipping. Adds this much latency.
fx.lim.att = Attack
fx.lim.att_tip = How soon the slow release stage takes over. Peaks shorter than this are left to the fast transient stage.
fx.lim.rel = Release
fx.lim.rel_tip = How the slow release stage lets go once an overload has passed. Longer is smoother; shorter is louder and can pump.
fx.lim.link_t = Link transients
fx.lim.link_t_tip = Stereo link of the fast stage. Below 100 % a peak on one side is caught on that side alone, so the image does not jump.
fx.lim.link_r = Link release
fx.lim.link_r_tip = Stereo link of the slow stage. Keep it high so the recovery stays centred.
fx.lim.knee = Knee
fx.lim.knee_tip = A soft knee: the limiting starts this many dB under the ceiling and bends in gently. The ceiling itself is never passed.
fx.lim.clip = Clipper
fx.lim.clip_tip = A clipper before the limiter, oversampled 4x. More of each peak is clipped and less is limited as it goes up; off at 0.
fx.lim.tp = True peak
fx.lim.tp_tip = Measure and hold the peaks between the samples, as a converter or a lossy encoder will see them. Adds 90 samples of latency (1.9 ms at 48 kHz).
fx.lim.dc = DC filter
fx.lim.dc_tip = Remove a DC offset (5 Hz high-pass) before limiting, so an offset does not make the limiting lopsided.
fx.lim.monitor = Hear
fx.lim.delta = Delta
fx.lim.delta_tip = Hear only what the limiter takes away. If you can hear the music in it, it is working too hard.
fx.lim.dither = Dither
fx.lim.dither_tip = TPDF dither for a lower word length. Only the last thing before the file should dither. Digital silence stays silent.
fx.lim.shape = Noise shaping
fx.lim.shape_tip = Where the dither's noise goes.
fx.lim.target = Target
