# Opciones
ds.mode.0 = Voz sola
ds.mode.0_tip = Una voz: un clasificador decide qué parte de la banda es sibilancia — la parte de la energía de la voz que cae en la banda, y cuánto se parece a ruido —, así que las vocales brillantes y los silbidos se respetan, sea cual sea su nivel
ds.mode.1 = General
ds.mode.1_tip = Todo lo que en la banda pase del umbral: mezclas, baterías, platillos, o un limitador de agudos
ds.proc.0 = Banda completa
ds.proc.0_tip = Baja toda la señal durante la sibilancia — lo más natural en una voz sola. Sin latencia si no hay anticipación
ds.proc.1 = Dividir agudos
ds.proc.1_tip = Baja solo lo que está por encima del borde bajo de la banda, con una división de fase lineal (2,6 ms de latencia)
ds.proc.2 = Dividir banda
ds.proc.2_tip = Baja solo la banda, del borde bajo al alto, y deja intacto el aire por encima (2,6 ms de latencia)
ds.stereo.0 = I/D
ds.stereo.0_tip = Izquierda y derecha, cada una con su detector, unidas por el enlace
ds.stereo.1 = M/S
ds.stereo.1_tip = Centro y lados, cada uno con su detector, unidos por el enlace
ds.stereo.2 = Centro
ds.stereo.2_tip = Solo se detecta y se baja el centro; los lados pasan intactos
ds.stereo.3 = Lados
ds.stereo.3_tip = Solo se detectan y se bajan los lados — coros paneados; el centro pasa intacto
ds.listen.0 = Salida
ds.listen.0_tip = La señal ya sin eses
ds.listen.1 = Disparo
ds.listen.1_tip = Lo que oye el detector: la clave a través de la banda de detección. Mueve la banda hasta oír solo la «s»
ds.listen.2 = Quitado
ds.listen.2_tip = Lo que se quita: entrada menos salida. Si en ello se oyen palabras, está trabajando de más
