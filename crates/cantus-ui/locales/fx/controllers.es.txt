# Presets (fxd step 3)
fxd.preset.xy.filter_sweep = Barrido de filtro

fxd.preset.xy.width_pan = Anchura y panorama
fxd.preset.xy.width_pan.what = Enlaza X a un panorama e Y a una anchura: X panoramiza a ambos lados, Y estrecha o ensancha en torno a las perillas
fxd.preset.xy.figure_eight = Figura de ocho
fxd.preset.xy.figure_eight.what = Un recorrido suave en forma de ocho de dos compases que suena con la canción: enlaza X e Y a dos parámetros cualesquiera
fxd.preset.xy.performance = Pad de interpretación
fxd.preset.xy.performance.what = A ambos lados del centro con un deslizamiento lento de 400 ms: enlaza X e Y a los dos parámetros que tocas en directo
fxd.preset.formula.sine = Seno
fxd.preset.formula.sine.what = Enlázala a un corte o a un panorama: una oscilación suave en torno al valor de la perilla; a la velocidad, b la fase, c la profundidad
fxd.preset.formula.triangle = Triángulo
fxd.preset.formula.triangle.what = Enlázala a un tono o a un panorama: sube y baja en línea recta en torno al valor de la perilla; a la velocidad, b la fase, c la profundidad
fxd.preset.formula.saw = Diente de sierra
fxd.preset.formula.saw.what = Enlázala a un corte para un barrido que sube y vuelve a caer en cada ciclo; a la velocidad, b la fase, c la profundidad
fxd.preset.formula.square = Cuadrada
fxd.preset.formula.square.what = Enlázala a un panorama o a un nivel: dos valores por turnos, medio ciclo cada uno; a la velocidad, b la fase, c la profundidad
fxd.preset.formula.pulse = Pulso
fxd.preset.formula.pulse.what = Enlázala a un nivel o a un envío: un alto breve y luego bajo; a la velocidad, b el ancho del pulso, c la profundidad
fxd.preset.formula.random_beat = Aleatorio por tiempo
fxd.preset.formula.random_beat.what = Enlázala a un corte o a un panorama: un valor aleatorio nuevo en cada tiempo, el mismo en cada render; a la velocidad, b otra secuencia, c la profundidad
fxd.preset.formula.gate_beats = Puerta por tiempos
fxd.preset.formula.gate_beats.what = Enlázala a un nivel (el fader abajo, profundidad positiva) o a un envío: se abre en cada tiempo durante el ancho b; a la velocidad, c la profundidad
fxd.preset.formula.env_bar = Envolvente por compás
fxd.preset.formula.env_bar.what = Enlázala a un corte o a un nivel: golpea en cada línea de compás y luego se apaga; a el ataque, b el decaimiento, c la profundidad
fxd.preset.formula.climb = Subida
fxd.preset.formula.climb.what = Enlázala a un corte o a un envío para una subida: crece desde el inicio de la canción a lo largo de a (de 1 a 64 compases), curvada por b, hasta la altura c
