# Area: external - ronda 36 (ADR 0094 unidades externas y ping, ADR 0095 Mackie Control y HUI). Cada clave necesita su gemela en external.en.txt

# --- el estado de una unidad ---
external.state.ready = lista
external.state.off = apagada
external.state.no_device = no hay dispositivo de audio abierto
external.state.other_interface = hay otra interfaz de audio abierta
external.state.channels = la interfaz no tiene esos canales
external.state.send_master = el envío sonaría en las salidas 1-2, la master: elige otro par
external.state.port = el puerto MIDI no está
external.state.track = su pista ya no existe

# --- el ping ---
external.ping = Ping
external.remeasure = Medir de nuevo
external.ping.running = midiendo...
external.ping.silent = no volvió nada: revisa el cable y los canales de retorno
external.ping.no_input = no hay entrada de audio abierta
external.ping.not_found = volvió algo, pero no el ping (zumbido u otra fuente)
external.ping.dropped = la entrada perdió audio al medir: inténtalo de nuevo
external.ping.unstable = las mediciones no coinciden: el tiempo de la interfaz no es estable
external.ping.no_note = el sintetizador no respondió a la nota: revisa el puerto, el canal y que el sonido suene
external.ping.busy = ya hay otra medición en marcha
external.ping.no_unit = la unidad ya no existe
external.ping.playing = detén primero el transporte
