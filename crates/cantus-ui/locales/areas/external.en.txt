# --- the state of a unit ---

external.state.channels = the interface has no such channels
external.state.send_master = the send would play on outputs 1-2, the master: choose another pair
external.state.port = the MIDI port is not there
external.state.track = its track is gone

# --- the ping ---
external.ping = Ping
external.remeasure = Measure again
external.ping.running = measuring...
external.ping.silent = nothing came back: check the cable and the return channels
external.ping.no_input = no audio input is open
external.ping.not_found = something came back, but not the ping (hum, or another source)
external.ping.dropped = the input lost audio while measuring: try again
external.ping.unstable = the measurements disagree: the interface's timing is not steady
external.ping.no_note = the synth did not answer the note: check the port, the channel and that the patch makes sound
external.ping.busy = another measurement is running
external.ping.no_unit = the unit is gone
external.ping.playing = stop the transport first
