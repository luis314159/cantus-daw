# --- why something did not work ---

update.error.held = The network answered with a web page instead of the file (a proxy may be scanning it). Try again in a few minutes.
update.error.rate_limited = GitHub is limiting requests from this network. Cantus will try again tomorrow.
update.error.answer = GitHub's answer was not a release Cantus can read.
update.error.hash = The download did not match its published SHA-256, so it was deleted.
update.error.no_hash = The release publishes no SHA-256 for this file, so Cantus will not install it.
update.error.signature = The installer is not signed by Cantus's publisher, so it was deleted.
update.error.io = The download could not be saved.
update.error.launch = The installer did not start: {error}
